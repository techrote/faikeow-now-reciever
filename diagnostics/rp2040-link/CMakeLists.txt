cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(fnr_link_diag C CXX ASM)
pico_sdk_init()
add_executable(fnr_link_diag main.c)
target_include_directories(fnr_link_diag PRIVATE ../common)
target_link_libraries(fnr_link_diag pico_stdlib hardware_uart)
target_compile_options(fnr_link_diag PRIVATE -Wall -Wextra -Werror)
pico_enable_stdio_usb(fnr_link_diag 1)
pico_enable_stdio_uart(fnr_link_diag 0)
pico_set_binary_type(fnr_link_diag no_flash)
pico_set_program_name(fnr_link_diag "FNR-002 UART link integrity diagnostic (RAM)")
pico_add_extra_outputs(fnr_link_diag)
