cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(fnr_observer C CXX ASM)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
add_executable(fnr_observer main.c)
target_link_libraries(fnr_observer pico_stdlib hardware_flash hardware_pio)
target_compile_options(fnr_observer PRIVATE -Wall -Wextra -Werror)
pico_enable_stdio_usb(fnr_observer 1)
pico_enable_stdio_uart(fnr_observer 0)
# Default build runs entirely in SRAM. It cannot replace installed flash.
pico_set_binary_type(fnr_observer no_flash)
pico_set_program_name(fnr_observer "FNR-002 high-impedance observer (RAM)")
pico_add_extra_outputs(fnr_observer)
