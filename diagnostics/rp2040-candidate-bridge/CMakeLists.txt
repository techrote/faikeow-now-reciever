cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(fnr_candidate_bridge C CXX ASM)
pico_sdk_init()
if(NOT FNR_ALLOW_CANDIDATE_UART)
  message(FATAL_ERROR "Explicit FNR_ALLOW_CANDIDATE_UART=1 required after passive observation")
endif()
add_executable(fnr_candidate_bridge main.c)
target_link_libraries(fnr_candidate_bridge pico_stdlib hardware_uart)
target_compile_options(fnr_candidate_bridge PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(fnr_candidate_bridge PRIVATE PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0)
pico_enable_stdio_usb(fnr_candidate_bridge 1)
pico_enable_stdio_uart(fnr_candidate_bridge 0)
pico_set_binary_type(fnr_candidate_bridge no_flash)
pico_set_program_name(fnr_candidate_bridge "FNR-002 candidate UART0 bridge (RAM)")
pico_add_extra_outputs(fnr_candidate_bridge)
