cmake_minimum_required(VERSION 3.20)
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(fnr_candidate_bridge C CXX ASM)
pico_sdk_init()
if(NOT FNR_ALLOW_CANDIDATE_UART)
  message(FATAL_ERROR "Explicit FNR_ALLOW_CANDIDATE_UART=1 required after passive observation")
endif()
add_executable(fnr_candidate_bridge main.c)
target_link_libraries(fnr_candidate_bridge pico_stdlib hardware_uart)
target_compile_options(fnr_candidate_bridge PRIVATE -Wall -Wextra -Werror)
target_compile_definitions(fnr_candidate_bridge PRIVATE
  PICO_STDIO_USB_ENABLE_RESET_VIA_BAUD_RATE=0
  PICO_STDIO_USB_ENABLE_IRQ_BACKGROUND_TASK=0
  PICO_STDIO_USB_CONNECTION_WITHOUT_DTR=1)
pico_enable_stdio_usb(fnr_candidate_bridge 1)
pico_enable_stdio_uart(fnr_candidate_bridge 0)
if(FNR_DIAG_BOARD_LABEL)
  if(NOT FNR_DIAG_BOARD_LABEL MATCHES "^[AB]$")
    message(FATAL_ERROR "FNR_DIAG_BOARD_LABEL must be A or B")
  endif()
  target_sources(fnr_candidate_bridge PRIVATE ../common/usb_board_label.c)
  target_compile_definitions(fnr_candidate_bridge PRIVATE FNR_DIAG_BOARD_LABEL="${FNR_DIAG_BOARD_LABEL}")
  target_link_options(fnr_candidate_bridge PRIVATE -Wl,--wrap=pico_get_unique_board_id_string)
endif()
if(FNR_BRIDGE_FLASH_RECOVERY_TEST)
  if(PICO_NO_FLASH OR NOT FNR_DIAG_BOARD_LABEL)
    message(FATAL_ERROR "Flash bridge requires PICO_NO_FLASH=0 and explicit board label")
  endif()
  pico_set_program_name(fnr_candidate_bridge "FNR-002 UART0 recovery bridge (FLASH)")
else()
  pico_set_binary_type(fnr_candidate_bridge no_flash)
  pico_set_program_name(fnr_candidate_bridge "FNR-002 candidate UART0 bridge (RAM)")
endif()
pico_add_extra_outputs(fnr_candidate_bridge)
