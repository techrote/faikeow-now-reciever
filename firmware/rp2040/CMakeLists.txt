cmake_minimum_required(VERSION 3.20)
if(NOT PICO_SDK_PATH)
  message(FATAL_ERROR "Pass the verified local -DPICO_SDK_PATH; fetching is not allowed")
endif()
set(PICO_BOARD pico CACHE STRING "Compile surrogate, NOT a clone pin contract")
set(PICO_PLATFORM rp2040 CACHE STRING "Target")
if(NOT PICO_BOARD STREQUAL "pico" OR NOT PICO_PLATFORM STREQUAL "rp2040")
  message(FATAL_ERROR "This prepass only defines the inert RP2040/pico compile surrogate")
endif()
set(PICO_TINYUSB_PATH "${PICO_SDK_PATH}/lib/tinyusb" CACHE PATH "Locked SDK gitlink" FORCE)
set(PICO_SDK_FETCH_FROM_GIT OFF CACHE BOOL "No implicit fetching" FORCE)
include("${PICO_SDK_PATH}/external/pico_sdk_import.cmake")
project(fnr_rp2040 LANGUAGES C CXX ASM)
set(CMAKE_C_STANDARD 17)
set(CMAKE_CXX_STANDARD 17)
pico_sdk_init()
if(NOT PICO_SDK_VERSION_STRING VERSION_EQUAL "2.2.0")
  message(FATAL_ERROR "Requires SDK 2.2.0 plus the exact git IDs in dependencies.lock.json")
endif()
find_package(picotool 2.2.0 REQUIRED CONFIG)
get_filename_component(FNR_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(fnr_rp2040_inert main.c usb_adapter.c ${FNR_ROOT}/shared/src/datagram.c)
target_include_directories(fnr_rp2040_inert PRIVATE . ${FNR_ROOT}/shared/include)
target_link_libraries(fnr_rp2040_inert PRIVATE pico_stdlib tinyusb_device)
pico_enable_stdio_usb(fnr_rp2040_inert 0)
pico_enable_stdio_uart(fnr_rp2040_inert 0)
pico_add_extra_outputs(fnr_rp2040_inert)
